cmake_minimum_required(VERSION 3.22.1)
project(pocket_diffusion LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    # Keep CPU kernels optimized even inside the development APK.
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Native inference build type" FORCE)
endif()
set(SD_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/stable-diffusion.cpp")
if(NOT EXISTS "${SD_SOURCE}/ggml/src/CMakeLists.txt")
    message(FATAL_ERROR "Run scripts/bootstrap-native.sh before building Pocket.")
endif()
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_GGML_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
# Android binaries must not inherit the build machine's CPU instructions.
if(ANDROID)
    set(GGML_AVX OFF CACHE BOOL "" FORCE)
    set(GGML_AVX2 OFF CACHE BOOL "" FORCE)
    set(GGML_FMA OFF CACHE BOOL "" FORCE)
    set(GGML_F16C OFF CACHE BOOL "" FORCE)
    set(GGML_BMI2 OFF CACHE BOOL "" FORCE)
else()
    # Optional host smoke tests use this machine's vector kernels.
    set(GGML_NATIVE ON CACHE BOOL "" FORCE)
    set(GGML_AVX ON CACHE BOOL "" FORCE)
    set(GGML_AVX2 ON CACHE BOOL "" FORCE)
    set(GGML_FMA ON CACHE BOOL "" FORCE)
    set(GGML_F16C ON CACHE BOOL "" FORCE)
endif()
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${SD_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/sdcpp" EXCLUDE_FROM_ALL)
add_library(pocket_diffusion SHARED diffusion_jni.cpp)
if(NOT ANDROID)
    find_path(JNI_HEADERS jni.h HINTS "$ENV{JAVA_HOME}/include" REQUIRED)
    find_path(JNI_PLATFORM_HEADERS jni_md.h HINTS "${JNI_HEADERS}/linux" "${JNI_HEADERS}/darwin" REQUIRED)
    target_include_directories(pocket_diffusion PRIVATE ${JNI_HEADERS} ${JNI_PLATFORM_HEADERS})
endif()
target_link_libraries(pocket_diffusion PRIVATE stable-diffusion)
target_link_options(pocket_diffusion PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
target_compile_options(pocket_diffusion PRIVATE -fvisibility=hidden)
