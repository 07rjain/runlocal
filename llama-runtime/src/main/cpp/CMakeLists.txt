cmake_minimum_required(VERSION 3.22.1)
project(pocket_llama LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Native inference build type" FORCE)
set(LLAMA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/llama.cpp")
if(NOT EXISTS "${LLAMA_SOURCE}/include/llama.h")
    message(FATAL_ERROR "Run scripts/bootstrap-native.sh before building the llama.cpp adapter.")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(pocket_llama SHARED llama_jni.cpp)
if(NOT ANDROID)
    find_path(JNI_HEADERS jni.h HINTS "$ENV{JAVA_HOME}/include" REQUIRED)
    find_path(JNI_PLATFORM_HEADERS jni_md.h HINTS "${JNI_HEADERS}/linux" "${JNI_HEADERS}/darwin" REQUIRED)
    target_include_directories(pocket_llama PRIVATE ${JNI_HEADERS} ${JNI_PLATFORM_HEADERS})
endif()
target_compile_options(pocket_llama PRIVATE -fvisibility=hidden)
target_link_libraries(pocket_llama PRIVATE llama)
target_link_options(pocket_llama PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
